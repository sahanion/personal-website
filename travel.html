Supabase Database Setup

Connect your free Supabase project to sync stories across devices. If left empty, stories will continue to be safely stored in your local browser store.

SQL to create table & secure RLS (click to expand) 
-- 1. Create travel stories table
create table if not exists travel_stories (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  location text not null,
  visit_date date not null,
  content text not null,
  photos jsonb default '[]'::jsonb,
  created_at timestamptz default now()
);

-- 2. Enable Row Level Security (RLS)
alter table travel_stories enable row level security;

-- 3. Public read policy: Anyone can view stories
create policy "Public can view stories"
  on travel_stories for select
  using (true);

-- 4. Author write policies: Only authenticated users can write/edit/delete
create policy "Authenticated author can insert"
  on travel_stories for insert
  with check (auth.role() = 'authenticated');

create policy "Authenticated author can update"
  on travel_stories for update
  using (auth.role() = 'authenticated');

create policy "Authenticated author can delete"
  on travel_stories for delete
  using (auth.role() = 'authenticated');
